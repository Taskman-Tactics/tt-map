 gdal_translate \
	-a_ullr -1024 750.2 1024 -750.2 \
 	-of COG \
	-co COMPRESS=JPEG \
  	-co TILING_SCHEME=GoogleMapsCompatible \
  	-co BLOCKSIZE=256 \
	-co QUALITY=75 \
	basemap.jpg basemap.tif

gdal raster pipeline read basemap.webp \
	! resize \
		-r nearest \
		--size 200%,0 \
	! edit \
		--crs "+proj=eqc +lat_ts=0 +lat_0=0 +lon_0=0 +x_0=0 +y_0=0 +datum=WGS84 +units=m +no_defs" \
		--bbox 0,0,4096,3006 \
	! write \
		--of COG \
		--co COMPRESS=JPEG \
		--co TILING_SCHEME=GoogleMapsCompatible \
		--co BLOCKSIZE=256 \
		--co QUALITY=100 \
		--co TARGET_SRS=EPSG:3857 \
		basemap.tif --overwrite
